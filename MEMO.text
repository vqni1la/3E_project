進化がうまくいかなければ、ステージ4か5でリセットする

操作一覧(マウスクリック操作可能){
  S           : センサーオンオフ
  M           : プレイヤーモード & 学習モードの切り替え
  R           : プレイヤーリセット
　ESC         : 終了

  1           : コース1 ただの四角
  2           : これ使わないで
  3           : コース3 ひし形
  4           : コース4 ぐねぐね
  5           : コース5 ぐねぐね
  6           : コース6 ちょい丸っぽい四角

  学習リセット : reset Learning
  時間速度変更 : ＋ - のボタンを押す
}

ラジコンキット(モーターとタイヤと車体) : Annadue 4WD
モーター制御 : TB6612FNG
マイコン : ESP32-WROOM
操作 : 差動駆動
センサー : 赤外線センサー

制御プログラムは下記のみ

#include <Bluepad32.h>
 
// --- ピン定義（画像の通りに設定） ---
 
// 上のTB6612（前輪用モジュール）
 
#define MOTOR_TOP_PWMA  18
 
#define MOTOR_TOP_AI1   15
 
#define MOTOR_TOP_AI2   4
 
#define MOTOR_TOP_PWMB  19
 
#define MOTOR_TOP_BI1   16
 
#define MOTOR_TOP_BI2   17
 
// 下のTB6612（後輪用モジュール）
 
#define MOTOR_BOT_PWMA  2
 
#define MOTOR_BOT_AIN1  21
 
#define MOTOR_BOT_AIN2  22
 
#define MOTOR_BOT_PWMB  25
 
#define MOTOR_BOT_BIN1  23
 
#define MOTOR_BOT_BIN2  5
 
ControllerPtr myControllers[BP32_MAX_GAMEPADS];
 
// モーター個別制御関数
 
void setMotor(int pwnPin, int in1Pin, int in2Pin, int speed) {
 
  if (speed > 0) {
 
    digitalWrite(in1Pin, HIGH);
 
    digitalWrite(in2Pin, LOW);
 
    analogWrite(pwnPin, speed);
 
  } else if (speed < 0) {
 
    digitalWrite(in1Pin, LOW);
 
    digitalWrite(in2Pin, HIGH);
 
    analogWrite(pwnPin, -speed);
 
  } else {
 
    digitalWrite(in1Pin, LOW);
 
    digitalWrite(in2Pin, LOW);
 
    analogWrite(pwnPin, 0);
 
  }
 
}
 
// 左右の車輪の速度を設定する関数 (-255 〜 255)
 
void driveCar(int leftSpeed, int rightSpeed) {
 
  // 左側モーター（上のAチャンネル + 下のAチャンネル）
 
  setMotor(MOTOR_TOP_PWMA, MOTOR_TOP_AI1, MOTOR_TOP_AI2, leftSpeed);
 
  setMotor(MOTOR_BOT_PWMA, MOTOR_BOT_AIN1, MOTOR_BOT_AIN2, leftSpeed);
 
  // 右側モーター（上のBチャンネル + 下のBチャンネル）
 
  setMotor(MOTOR_TOP_PWMB, MOTOR_TOP_BI1, MOTOR_TOP_BI2, rightSpeed);
 
  setMotor(MOTOR_BOT_PWMB, MOTOR_BOT_BIN1, MOTOR_BOT_BIN2, rightSpeed);
 
}
 
// プロコンが接続されたとき
 
void onConnectedController(ControllerPtr ctl) {
 
  bool foundEmptySlot = false;
 
  for (int i = 0; i < BP32_MAX_GAMEPADS; i++) {
 
    if (myControllers[i] == nullptr) {
 
      myControllers[i] = ctl;
 
      foundEmptySlot = true;
 
      Serial.printf("コントローラーが接続されました: INDEX=%d\n", i);
 
      break;
 
    }
 
  }
 
}
 
// プロコンが切断されたとき
 
void onDisconnectedController(ControllerPtr ctl) {
 
  for (int i = 0; i < BP32_MAX_GAMEPADS; i++) {
 
    if (myControllers[i] == ctl) {
 
      myControllers[i] = nullptr;
 
      Serial.printf("コントローラーが切断されました: INDEX=%d\n", i);
 
      driveCar(0, 0); // 停止
 
      break;
 
    }
 
  }
 
}
 
// コントローラーの操作処理
 
void processGamepad(ControllerPtr ctl) {
 
  // スティックの値を取得 (-512 〜 511)
 
  int axisY = ctl->axisY(); // 左スティック上下（前後進）
 
  int axisRX = ctl->axisRX(); // 右スティック左右（旋回）
 
  // デッドゾーン処理（スティックの微小なブレを無視）
 
  if (abs(axisY) < 50) axisY = 0;
 
  if (abs(axisRX) < 50) axisRX = 0;
 
  // 速度調整 (-255 〜 255 に変換)
 
  // ※ Y軸は上がマイナス値になるため反転
 
  int throttle = map(-axisY, -512, 512, -255, 255);
 
  int steering = map(axisRX, -512, 512, -255, 255);
 
  // 左右の速度計算（スキッドステア方式）
 
  int leftMotorSpeed = throttle + steering;
 
  int rightMotorSpeed = throttle - steering;
 
  // 範囲制限 (-255 〜 255)
 
  leftMotorSpeed = constrain(leftMotorSpeed, -255, 255);
 
  rightMotorSpeed = constrain(rightMotorSpeed, -255, 255);
 
  // 車輪へ出力
 
  driveCar(leftMotorSpeed, rightMotorSpeed);
 
}
 
void setup() {
 
  Serial.begin(115200);
 
  // モーターピンを出力モードに設定
 
  pinMode(MOTOR_TOP_PWMA, OUTPUT);
 
  pinMode(MOTOR_TOP_AI1, OUTPUT);
 
  pinMode(MOTOR_TOP_AI2, OUTPUT);
 
  pinMode(MOTOR_TOP_PWMB, OUTPUT);
 
  pinMode(MOTOR_TOP_BI1, OUTPUT);
 
  pinMode(MOTOR_TOP_BI2, OUTPUT);
 
  pinMode(MOTOR_BOT_PWMA, OUTPUT);
 
  pinMode(MOTOR_BOT_AIN1, OUTPUT);
 
  pinMode(MOTOR_BOT_AIN2, OUTPUT);
 
  pinMode(MOTOR_BOT_PWMB, OUTPUT);
 
  pinMode(MOTOR_BOT_BIN1, OUTPUT);
 
  pinMode(MOTOR_BOT_BIN2, OUTPUT);
 
  // Bluepad32の初期化
 
  BP32.setup(&onConnectedController, &onDisconnectedController);
 
  BP32.forgetBluetoothKeys(); // 過去のペアリング情報をクリアして接続しやすくする
 
}
 
void loop() {
 
  BP32.update();
 
  for (int i = 0; i < BP32_MAX_GAMEPADS; i++) {
 
    ControllerPtr myCtl = myControllers[i];
 
    if (myCtl && myCtl->isConnected()) {
 
      if (myCtl->isGamepad()) {
 
        processGamepad(myCtl);
 
      }
 
    }
 
  }
 
  delay(15);
 
}
